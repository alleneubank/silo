const pids = new Set<number>();
const objects = new Map<string, {kind:string;id:string;name:string}>();
function object(kind:string,id:string,name=id) {objects.set(`${kind}:${id}`,{kind,id,name});}
function receipt(r:any) {
  object(r.kind,r.id,r.name);
  if(r.kind==='container') for(const v of r.anonymousVolumes??[])object('volume',v.name);
}
function dockerInspection(row:any) {
  if(!row || typeof row!=='object')return;
  if(row.Id&&row.Name&&row.Created&&row.State)object('container',row.Id,row.Name.replace(/^\//,''));
  else if(row.Id&&row.Name&&row.CreatedAt===undefined&&row.Containers!==undefined)object('network',row.Id,row.Name);
  else if(row.Name&&row.CreatedAt&&row.Mountpoint)object('volume',row.Name);
  for(const m of row.Mounts??[])if(m.Type==='volume'&&m.Name)object('volume',m.Name);
  if(row.Type==='volume'&&row.Name)object('volume',row.Name);
}
function scan(directory:string) {
  if(!existsSync(directory))return;
  for(const e of readdirSync(directory,{withFileTypes:true})) {
    const file=`${directory}/${e.name}`;
    if(e.isDirectory())scan(file);
    else if(e.name==='resources'||e.name==='resources.txt') {
      const text=readFileSync(file,'utf8');
      for(const m of text.matchAll(/(?:\bprocess |pid\/pgid |supervisor=|group=)(\d+)/g))pids.add(Number(m[1]));
      for(const m of text.matchAll(/^docker (container )?rm(?: -\S+)* ([a-f0-9]{64})/gm))object('container',m[2]);
      for(const m of text.matchAll(/^docker (volume|network) rm (\S+)/gm))object(m[1],m[2]);
      for(const m of text.matchAll(/^(network|volume) (\S+) name=(\S+);/gm))object(m[1],m[2],m[3]);
      for(const m of text.matchAll(/^(?:synthetic stopped k3d (?:server|registry) container|outside-claim synthetic container|donor container|borrower container) ([a-f0-9]{64})/gm))object('container',m[1]);
    } else if(e.name==='recorded-process-pids.txt') {
      for(const text of readFileSync(file,'utf8').trim().split(/\s+/))pids.add(Number(text));
    } else if(e.name.endsWith('.stdout')) {
      let value:any;try{value=JSON.parse(readFileSync(file,'utf8'));}catch{continue;}
      if(value?.project)for(const r of value.project.resources??[])receipt(r);
      if(Array.isArray(value))for(const row of value)dockerInspection(row);
    }
  }
}
