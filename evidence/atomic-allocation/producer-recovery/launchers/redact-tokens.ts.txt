import{readFileSync,writeFileSync,readdirSync,openSync,ftruncateSync,closeSync}from'node:fs';import path from'node:path';
export const marker='[REDACTED_EPHEMERAL_FIXTURE_TOKEN]';
export function redact(s:string){return s.replace(/K3S_TOKEN=[^"\\\s]+/g,'K3S_TOKEN='+marker).replace(/("k3d\.cluster\.token"\s*:\s*")[^"]*(")/g,'$1'+marker+'$2');}
function tree(dir:string){for(const e of readdirSync(dir,{withFileTypes:true})){const f=path.join(dir,e.name);if(e.isDirectory()){if(!['account','cache','tmp','bin','hook-push.git','legacy','full-example'].includes(e.name))tree(f);}else if(e.isFile()&&(/\.(json|jsonl|stdout|stderr|log|out|err)$/.test(f)||['resources','ledger'].includes(e.name))){const before=readFileSync(f,'utf8'),after=redact(before);if(before!==after)writeFileSync(f,after);}}}
if(import.meta.main){if(process.argv[2]==='all')tree(process.argv[3]);else if(process.argv[2]==='ledger'){const file=process.argv[3],key=process.argv[4],all=readFileSync(file,'utf8'),at=all.indexOf(key);if(at<0)throw Error('Missing r7 marker');const after=all.slice(0,at)+redact(all.slice(at));const fd=openSync(file,'r+');try{writeFileSync(fd,after);ftruncateSync(fd,Buffer.byteLength(after));}finally{closeSync(fd);}}}
