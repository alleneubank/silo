import { Database } from "bun:sqlite";
import { existsSync, writeFileSync } from "node:fs";

const marker = "/tmp/completion-blocked.json";
const release = "/tmp/completion-release";
if (Bun.argv.includes("__silo_execute")) {
  const query = Database.prototype.query;
  Database.prototype.query = function (...args) {
    const statement = query.apply(this, args);
    if (String(args[0]).startsWith("INSERT INTO completions") && Bun.argv[Bun.argv.indexOf("__silo_execute") + 2]?.endsWith("/a")) {
      const run = statement.run;
      statement.run = function (...params) {
        const receipt = JSON.parse(String(params[3]));
        writeFileSync(marker, JSON.stringify(receipt));
        const deadline = Date.now() + 4000;
        while (!existsSync(release)) {
          if (Date.now() >= deadline) throw new Error("Completion scheduling probe was not released");
          Atomics.wait(new Int32Array(new SharedArrayBuffer(4)), 0, 0, 10);
        }
        return run.apply(this, params);
      };
    }
    return statement;
  };
} else {
  const spawn = Bun.spawn;
  Bun.spawn = function (argv, options) {
    if (Array.isArray(argv) && argv.includes("__silo_execute")) {
      const executor = argv.indexOf("__silo_execute");
      argv = [...argv.slice(0, executor - 1), "--preload", "/tmp/hold-completion.ts", ...argv.slice(executor - 1)];
    } else if (Array.isArray(argv) && argv[0] === process.execPath && argv[1] === "/mnt/dist/cli.js" && argv.includes("up")) {
      argv = [argv[0], "--preload", "/tmp/hold-completion.ts", ...argv.slice(1)];
    }
    return spawn(argv, options);
  };
}
