function visit(root: string) {
  for (const entry of readdirSync(root, { withFileTypes: true })) {
    const file = root + "/" + entry.name;
    if (entry.isDirectory()) visit(file);
    else if (entry.name === "resources") {

      for (const match of text.matchAll(/(?:\bprocess |pid\/pgid |supervisor=|group=)(\d+)/g)) handles.add(Number(match[1]));
    } else if (entry.name === "server.pid" || root === "/tmp/pids") handles.add(Number(readFileSync(file, "utf8").trim()));
  }
}

async function docker(args: string[]) {
  const proc = Bun.spawn(["docker", ...args], { stdout: "pipe", stderr: "pipe" });
  const [stdout, stderr, code] = await Promise.all([new Response(proc.stdout).text(), new Response(proc.stderr).text(), proc.exited]);
  assert.equal(code, 0, stderr); return stdout.trim();
}
